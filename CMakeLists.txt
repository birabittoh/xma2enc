cmake_minimum_required(VERSION 3.16)
project(xma2enc CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(xma2enc STATIC src/encoder.cpp)
target_include_directories(xma2enc PUBLIC include)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  add_executable(xma2enc_cli tools/xma2enc_cli.cpp)
  target_link_libraries(xma2enc_cli xma2enc)
endif()
